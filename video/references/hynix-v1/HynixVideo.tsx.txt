import React from 'react';
import {AbsoluteFill, Html5Audio, Img, Sequence, interpolate, staticFile, useCurrentFrame} from 'remotion';
import timeline from '../scenes/hynix.json';

const navy = '#122b42';
const teal = '#137e76';
const font = '"Apple SD Gothic Neo", "Noto Sans KR", sans-serif';
type Scene = typeof timeline.scenes[number];
const titles = ['실적으로 말하는 주식', '2분기, 얼마나 벌었나', '2분기 매출은?', '그리고, 3분기 소식', '보유자도 비중 확대'];
const notes = ['SK하이닉스 실적 브리핑', '2026년 2분기 · 영업 실적', '2026년 2분기 · 매출', '2026년 3분기 · 영업이익', '실적으로 말하는 주식'];
const ease = (f: number, a=0, b=18) => interpolate(f,[a,b],[0,1],{extrapolateLeft:'clamp',extrapolateRight:'clamp'});

const Card: React.FC<{children:React.ReactNode; height?:number}> = ({children,height=440}) => <div style={{
  position:'absolute',left:64,top:386,width:952,height,borderRadius:32,
  background:'rgba(255,255,255,.97)',boxShadow:'0 16px 48px rgba(18,43,66,.12)',
  border:'2px solid rgba(18,43,66,.07)',padding:'40px 42px',overflow:'hidden',
}}>{children}</div>;

const Profit: React.FC<{frame:number}> = ({frame}) => {
  const p=ease(frame,12,60);
  return <Card><div style={{fontSize:31,color:'#58717e',fontWeight:650}}>영업이익</div>
    <div style={{display:'flex',alignItems:'center',justifyContent:'space-between',marginTop:18}}>
      <div><div style={{fontSize:136,lineHeight:1.15,fontWeight:850,letterSpacing:-6,color:navy}}>{Math.round(60*p)}<span style={{fontSize:65,marginLeft:10}}>조 원+</span></div>
        <div style={{marginTop:28,fontSize:30,color:'#58717e'}}>2026년 2분기</div></div>

        <svg width="250" height="250" viewBox="0 0 250 250"><circle cx="125" cy="125" r="103" fill="none" stroke="#e8f0ee" strokeWidth="23"/>
          <circle cx="125" cy="125" r="103" fill="none" stroke={teal} strokeWidth="23" strokeLinecap="round" strokeDasharray={`${647*.76*p} 647`} transform="rotate(-90 125 125)"/></svg>
        <div style={{position:'absolute',inset:0,display:'flex',flexDirection:'column',alignItems:'center',justifyContent:'center'}}>
          <div style={{fontSize:58,fontWeight:850,color:teal}}>{Math.round(76*p)}%</div><div style={{fontSize:25,color:'#58717e'}}>영업이익률</div></div>
      </div>
    </div></Card>;
};

const Revenue: React.FC<{frame:number}> = ({frame}) => {
  const p=ease(frame,8,58);
  return <Card height={480}><div style={{fontSize:31,color:'#58717e',fontWeight:650}}>분기 매출</div>

const Forecast: React.FC<{frame:number;duration:number}> = ({frame,duration}) => {
  const stampAt=Math.min(100,Math.floor(duration*.62));
  const stamp=ease(frame,stampAt,stampAt+7);
  return <Card height={440}><div style={{fontSize:31,color:'#58717e',fontWeight:650}}>3분기 영업이익</div>
    <div style={{fontSize:154,fontWeight:850,letterSpacing:-7,marginTop:20,lineHeight:1.2,color:navy}}>77<span style={{fontSize:71,marginLeft:16}}>조 원</span></div>
    <div style={{position:'absolute',right:52,bottom:48,border:'8px solid #b94643',borderRadius:12,
      color:'#b94643',fontSize:64,fontWeight:900,padding:'5px 25px',opacity:stamp,
      transform:`rotate(-9deg) scale(${interpolate(stamp,[0,1],[1.5,1])})`}}>확정</div>
    <div style={{fontSize:29,color:'#58717e',marginTop:29}}>3분기 실적 브리핑</div>
  </Card>;
};

const SceneView: React.FC<{scene:Scene}> = ({scene}) => {
  const f=useCurrentFrame();
  const enter=ease(f,0,12);
  const leave=interpolate(f,[scene.durationInFrames-6,scene.durationInFrames],[1,0],{extrapolateLeft:'clamp',extrapolateRight:'clamp'});
  const active=scene.subtitles.find(s=>f>=s.from && f<s.to)
    ?? (f<scene.subtitles[0].from ? scene.subtitles[0] : scene.subtitles[scene.subtitles.length-1]);
  const zoom=interpolate(f,[0,scene.durationInFrames],[1,1.025],{extrapolateRight:'clamp'});
  return <AbsoluteFill style={{opacity:Math.min(enter,leave),transform:`translateY(${(1-enter)*14}px)`}}>
    <div style={{position:'absolute',left:64,top:215,fontSize:58,fontWeight:850,letterSpacing:-2,color:navy}}>{titles[scene.id]}</div>
    <div style={{position:'absolute',left:66,top:304,fontSize:31,fontWeight:600,color:'#4e6a7a'}}>{notes[scene.id]}</div>
    {scene.id===0 && <Card height={410}>
      <div style={{fontSize:30,letterSpacing:3,fontWeight:650,color:teal}}>SEMICONDUCTOR BRIEF</div>
      <div style={{marginTop:30,fontSize:106,fontWeight:850,letterSpacing:-5,lineHeight:1.12,color:navy}}>SK하이닉스</div>
      <div style={{marginTop:28,fontSize:46,fontWeight:700,color:'#47616e'}}>숫자로 보겠습니다.</div>
      <div style={{height:7,width:118,background:teal,borderRadius:10,marginTop:32}}/>
    </Card>}
    {scene.id===1 && <Profit frame={f}/>}
    {scene.id===2 && <Revenue frame={f}/>}
    {scene.id===3 && <Forecast frame={f} duration={scene.durationInFrames}/>}
    {scene.id===4 && <Card height={410}>
      <div style={{fontSize:30,color:teal,fontWeight:700}}>이미 보유하고 있다면?</div>
      <div style={{fontSize:78,fontWeight:850,color:navy,lineHeight:1.3,letterSpacing:-3,marginTop:26}}>지금은<br/>비중을 늘릴 때</div>
      <div style={{position:'absolute',right:48,bottom:48,width:100,height:100,borderRadius:50,background:'#e3f1ed',color:teal,fontSize:72,textAlign:'center',lineHeight:'100px'}}>↗</div>
    </Card>}
    <div style={{position:'absolute',left:-22,top:scene.id===2?886:830,width:1124,height:1027,
      transform:`scale(${zoom})`,transformOrigin:'50% 80%'}}>
      <Img src={staticFile('assets/characters/chaewon-base-v1.png')} style={{width:'100%',height:'100%',objectFit:'contain'}}/>
    </div>
    <div style={{position:'absolute',left:66,top:1386,padding:'12px 23px',background:teal,borderRadius:14,
      color:'white',fontSize:29,fontWeight:750}}>이채원</div>
    <div style={{position:'absolute',left:64,top:1478,width:952,height:240,padding:'25px 38px',
      borderRadius:28,background:'rgba(14,35,54,.98)',boxShadow:'0 12px 40px #0c213633'}}>
      <div style={{fontSize:25,color:'#94c7c3',fontWeight:650,letterSpacing:1}}>실적으로 말하는 주식</div>
      <div style={{height:150,display:'flex',flexDirection:'column',justifyContent:'center',fontSize:49,fontWeight:760,
        lineHeight:1.43,color:'white',letterSpacing:-1.2}}>
        {active.lines.map((line,i)=><div key={i}>{line}</div>)}
      </div>
    </div>
    <div style={{position:'absolute',right:70,top:1396,color:navy,fontSize:27,fontWeight:750,
      background:'rgba(255,255,255,.94)',borderRadius:14,padding:'9px 16px'}}>{String(scene.id+1).padStart(2,'0')} / 05</div>
  </AbsoluteFill>;
};
